## Import

```tsx
import { SegmentedControl, SegmentedControlItem } from "@sigil-ui/components";
```

## Usage

```tsx
"use client";

import { useState } from "react";
import { SegmentedControl, SegmentedControlItem } from "@sigil-ui/components";

function SegmentedControlPreview() {
  const [period, setPeriod] = useState("Week");
  return <div className="grid gap-[var(--s-space-16)] w-full">
    <SegmentedControl value={period} onValueChange={setPeriod} aria-label="Report period">
      {["Day", "Week", "Month"].map((value) => <SegmentedControlItem key={value} value={value}>{value}</SegmentedControlItem>)}
    </SegmentedControl>
    <p role="status" className="text-[var(--s-text-muted)] text-[length:var(--s-size-sm)]">Showing the {period.toLowerCase()} report.</p>
  </div>;
}
```

## Props

| Prop | Type | Default | Description |
|---|---|---|---|
| `value` (required) | `string` | — | The current controlled value. |
| `onValueChange` (required) | `(value: string) => void` | — | Called when the user changes the value. |
| `children` (required) | `ReactNode` | — | Content rendered inside the component. |
