
## Usage

```tsx
<SocialIcons
    links={[
      { label: "GitHub", href: "https://github.com", icon: <span aria-hidden className="text-xs font-semibold">GH</span> },
      { label: "X", href: "https://x.com", icon: <span aria-hidden className="text-xs font-semibold">X</span> },
    ]}
  />
```

## Props

| Prop | Type | Default | Description |
|---|---|---|---|
| `links` (required) | `SocialLink[]` | — | Links configuration. |
| `size` | `"sm" \| "md"` | — | Sets the component's size. |
