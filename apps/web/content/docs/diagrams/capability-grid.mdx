## Import

```tsx
import { CapabilityGrid } from "@sigil-ui/components";
```

## Usage

```tsx
<CapabilityGrid
    categories={[
      { label: "Runtimes", items: [{ name: "Node" }, { name: "Python" }, { name: "Rust" }] },
      { label: "System", items: [{ name: "Root", color: "var(--s-success)" }, { name: "GPU", color: "var(--s-success)" }] },
    ]}
    callout={{ label: "Unavailable", items: ["Nested virtualization"], variant: "warning" }}
  />
```

## Props

| Prop | Type | Default | Description |
|---|---|---|---|
| `categories` (required) | `CapabilityCategory[]` | — | Categories configuration. |
| `callout` | `CapabilityCallout` | — | Callout configuration. |
